# Core Python dependencies. Works from a plain clone: no bqskit/ or bqskit-ft/
# checkout is needed for the Rust pipeline, the plotting scripts, or the
# cyclosynth/dcsynth converters. For the bqskit backend of compile_cliffordt.py
# use requirements-bqskit.txt instead (it includes this file).
numpy
matplotlib
pandas
qiskit>=2.5
# Needed for load_circuit's OpenQASM 3 fallback in data_processing/compile_cliffordt.py
# (e.g. MQT Bench's OQASM3 exports) -- qasm3.load() raises MissingOptionalLibraryError
# without it, which load_circuit's except-clause otherwise just folds into an opaque
# "could not parse as OpenQASM 2 (...) or OpenQASM 3 (...)" error.
qiskit-qasm3-import
# Optional: fallback rotation synthesiser for data_processing/compile_cliffordt.py,
# used only for the angles qiskit's Rust gridsynth panics on at coarse --epsilon.
# Also pulled in by bqskit-ft; listed here so a plain clone has it too.
pygridsynth
# Rz synthesis backend for data_processing/dcsynth_to_cliffordt.py (needs mpmath).
mpmath
# Type checker used to verify data_processing/compile_cliffordt.py during
# development (pyright data_processing/compile_cliffordt.py). Not a runtime
# dependency of the pipeline itself.
pyright
# Without this, pyright falls back to pandas' own (much less precise) inline
# stubs and misreports dozens of DataFrame/Series methods (dropna, apply,
# to_numpy, fillna, ...) as attribute-access errors across data_processing/
# (e.g. plot_puremagic.py). Not a runtime dependency.
pandas-stubs
